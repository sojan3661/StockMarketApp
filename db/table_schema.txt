Transactions
create table public."Transactions" (
  id bigint generated by default as identity not null,
  "Symbol" text not null,
  "BuyDate" date null,
  "SellDate" date null,
  "Qty" double precision null,
  "BuyAvg" double precision null,
  "SellAvg" double precision null,
  "Portfolio" text null,
  "BuyAvgToLocal" double precision null,
  "BuyAvgUSD" double precision null,
  "SellAvgLocal" double precision null,
  "SellAvgUSD" double precision null,
  "BuyValue" double precision null,
  "BuyValueUSD" double precision null,
  "SellValue" double precision null,
  "SellValueUSD" double precision null,
  constraint Transactions_pkey primary key (id),
  constraint Transactions_Portfolio_fkey foreign KEY ("Portfolio") references "Investment Plan" ("Portfolio"),
  constraint Transactions_Symbol_fkey foreign KEY ("Symbol") references "StockManagement" ("Symbol")
) TABLESPACE pg_default;

StockManagement
create table public."StockManagement" (
  "Symbol" text not null,
  "Name" text not null,
  "Equity" boolean null,
  "Sector" text null,
  "Listed" boolean null,
  "MarketCap" text null,
  "LTP" double precision null,
  "Country" text null default 'INDIA'::text,
  constraint StockManagement_pkey primary key ("Symbol"),
  constraint StockManagement_Sector_fkey foreign KEY ("Sector") references "Sectors" ("Sector")
) TABLESPACE pg_default;

StockAllocation
create table public."StockAllocation" (
  "Symbol" text not null,
  "Portfolio" text null,
  "Allocation" double precision null,
  id smallint generated by default as identity not null,
  constraint StockAllocation_pkey primary key (id),
  constraint StockAllocation_Portfolio_fkey foreign KEY ("Portfolio") references "Investment Plan" ("Portfolio"),
  constraint StockAllocation_Symbol_fkey foreign KEY ("Symbol") references "StockManagement" ("Symbol")
) TABLESPACE pg_default;

Sectors
create table public."Sectors" (
  "Sector" text not null,
  constraint Sectors_pkey primary key ("Sector")
) TABLESPACE pg_default;

SectorAllocation
create table public."SectorAllocation" (
  id bigint generated by default as identity not null,
  "Sector" text not null,
  "Allocation" double precision not null,
  "Portfolio" text null,
  constraint SectorAllocation_pkey primary key (id),
  constraint SectorAllocation_Portfolio_fkey foreign KEY ("Portfolio") references "Investment Plan" ("Portfolio"),
  constraint SectorAllocation_Sector_fkey foreign KEY ("Sector") references "Sectors" ("Sector")
) TABLESPACE pg_default;

Investment Plan
create table public."Investment Plan" (
  "Current Invested Amount" double precision not null,
  "Monthly SIP" double precision null,
  "Number of Months" smallint null,
  "Portfolio" text not null,
  "Description" text null,
  "Platform" text null,
  constraint Investment Plan_pkey primary key ("Portfolio")
) TABLESPACE pg_default;

  "CashEnd" double precision null,
